import sys, io, json
from PIL import Image

path = sys.argv[1] if len(sys.argv) > 1 else "downloaded.png"
img = Image.open(path)
info = img.info or {}
print("chunks:", list(info.keys()))
if "prompt" in info:
    p = json.loads(info["prompt"])
    print("prompt workflow 节点数:", len(p))
    for nid, node in p.items():
        print(" ", nid, node.get("class_type"))
